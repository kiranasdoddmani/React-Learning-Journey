       import { useEffect, useState } from "react"

function counter(){
    let[count,setCount]=useState(0);

    let CountChange=()=>{
        setCount((count)=>{
           return count+1;
        })
    }

  useEffect(function sideeffect() {
    console.log("this is a side-effect");
});

    return (
        <>
         <h1>Counter</h1>
             <p>count={count}</p>
             <button onClick={CountChange}>+1</button>
        </>
    )
}

export default counter;

              Using API Calls      

               useEffect(() => {
                  getNewJoke();

       Calling getNewJoke function and [] empty dependecies

       we use empty dependencies [] in useEffect() when we want the code inside useEffect
